# Van clavadas a una versión, y no con >= como las de runtime, por una
# razón concreta: una versión nueva de pylint estrena comprobaciones, y
# una comprobación nueva baja la puntuación de un código que no ha
# cambiado. Con el listón en 7.5, eso convierte una actualización
# silenciosa en un fallo que no lo es. Subirlas es una decisión, no algo
# que pase solo.
pytest==9.1.1
pylint==4.0.6
# PyInstaller tiene el mismo problema y peor: los comentarios de
# translator.spec explican que torch y tkinterdnd2 se dejan a sus hooks,
# y lo que esos hooks recogen cambia de version en version. Un build que
# se instala solo es un build que nadie eligio.
pyinstaller==6.22.1
